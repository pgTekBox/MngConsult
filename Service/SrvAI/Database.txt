ALTER procedure [dbo].[s1549InsertMail_A] @Message varchar(max),@RCPT varchar(1000),@From varchar(300),@BCC varchar(300) ,
	@CC varchar(300) ,	@ReplyTo varchar(300) ,	@ResentBCC varchar(300) ,	@ResentCC varchar(300) ,	@ResentFrom varchar(300) ,
	@ResentReplyTo varchar(300) ,	@ResentSender varchar(300) ,	@ResentTo varchar(300) ,	@Sender varchar(300) ,	@To varchar(300) ,
	@InReplyTo varchar(300) ,	@Importance varchar(20) ,	@xPriority varchar(20) ,	@MessageId varchar(300) ,	@ResentMessageId varchar(300) ,
	@Subject varchar(2000) ,	@TextBody varchar(max) ,	@HTMLBody varchar(max),@ClientIP varchar(50)  as


declare @ToSplited varchar(200)
DECLARE @EmailAdress varchar(300)
declare @FolderId int
declare @UserIdentification as uniqueidentifier  
 declare @NewMailId int
 

DECLARE @TBLMailId TABLE
(
  MailID int
)
-- -----------------------------------------------------------------------------------------------


--declare @RCPT varchar(3000)
--set @RCPT = ',<recrutement@sourcevolution.ca>,yperron@sourcevolution.ca,<tpilote@sourcevolution.ca>,<tpilote@sourcevolution.ca>,ventes@sourcevolution.ca'

declare @TBLMailAdresseRCPT TABLE
(
	EMail varchar(300)
)


DECLARE adresse_cursor CURSOR FOR   
     SELECT item  from dbo.fnSplit2(@RCPT,',')   
OPEN adresse_cursor  
FETCH NEXT FROM adresse_cursor   
                   INTO @EmailAdress 
WHILE @@FETCH_STATUS = 0  
BEGIN  
    set   @EmailAdress =  replace(@EmailAdress,'>','')
    set   @EmailAdress =  replace(@EmailAdress,'<','')


    if ((select count(*) from @TBLMailAdresseRCPT where EMail = @EmailAdress) = 0 )
	  begin
	    if (( select count(*) from [MTL-GI-05].[GestionInterne].[dbo].[T022Groups] where coalesce(Grouptype,0) = 1  and Email =   @EmailAdress   )  = 1 ) 
		      begin
			  --ici c est une groupe de mail
			    
			    insert into @TBLMailAdresseRCPT (Email)  (select Email from dbo.GetuserMailByGroupId(@EmailAdress) where email not in ( select Email from @TBLMailAdresseRCPT))

			  end
			  else
			  begin 
			      -- ici c est un user
				  insert into @TBLMailAdresseRCPT (Email)  (select @EmailAdress where @EmailAdress not in ( select Email from @TBLMailAdresseRCPT))
   			  end
      end



    FETCH NEXT FROM adresse_cursor   
    INTO @EmailAdress
END   
CLOSE adresse_cursor;  
DEALLOCATE adresse_cursor; 


-------------------------------------------------------------



  

DECLARE mail_cursor CURSOR FOR   
     SELECT '<' + EMail + '>' from @TBLMailAdresseRCPT  

OPEN mail_cursor  

FETCH NEXT FROM mail_cursor   
                   INTO @EmailAdress 

WHILE @@FETCH_STATUS = 0  
BEGIN  
      
      if (( select count(*) from [MTL-GI-05].[GestionInterne].[dbo].[T021Users] where '<' + [MailBox] + '>' =   @EmailAdress   )  = 1 ) 
	  begin 
	       select @UserIdentification  = [UserIdentification] from [MTL-GI-05].[GestionInterne].[dbo].[T021Users] where  '<' + [MailBox] + '>' =     @EmailAdress  
	       select @FolderId = Id from [dbo].[T401MailFolders] where  [UserId] = @UserIdentification and [FolderType] = 1 and ParentId is null
	  end 
	  else
	       set  @FolderId  = -1


      insert into [dbo].[T400Mails] ([Mail],[RCPT],RCPT_ORG,[Received],[From]  ,BCC  ,	CC ,	ReplyTo   ,	ResentBCC   ,	ResentCC   ,	ResentFrom   ,
	                            ResentReplyTo   ,	ResentSender  ,	ResentTo   ,	Sender  ,	[To]   ,	InReplyTo ,	Importance   ,
	                            xPriority   ,	MessageId   ,	ResentMessageId   ,	Subject  ,	TextBody ,	HTMLBody    ,ClientIP,FolderId,HasBeenNotifie,SMTPMail
                ) values (@Message, @EmailAdress,@RCPT, getdate(), @From  ,@BCC  ,	@CC ,	@ReplyTo   ,	@ResentBCC   ,	@ResentCC   ,	@ResentFrom   ,
	                            @ResentReplyTo   ,	@ResentSender  ,	@ResentTo   ,	@Sender  ,	@To   ,	@InReplyTo ,	@Importance   ,
	                            @xPriority   ,	@MessageId   ,	@ResentMessageId   ,	@Subject  ,	@TextBody ,	@HTMLBody,@ClientIP,@FolderId,0,1 )
    
	   set @NewMailId = @@IDENTITY 
	   insert into @TBLMailId (MailId) values(@NewMailId)
	   

        -- Get the next vendor.  
    FETCH NEXT FROM mail_cursor   
    INTO @EmailAdress
END   
CLOSE mail_cursor;  
DEALLOCATE mail_cursor;  

select MailId from @TBLMailId
